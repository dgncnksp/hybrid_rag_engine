import numpy as np
from typing import List
from sentence_transformers import SentenceTransformer

class DenseEmbedder:
    """
    Metinleri vektör uzayına taşıyan Dense Embedding sınıfı.
    Hafif ve etkili multilingual bir model (all-MiniLM-L6-v2 veya bge-m3 benzeri) kullanır.
    """
    def __init__(self, model_name: str = "sentence-transformers/all-MiniLM-L6-v2"):
        self.model = SentenceTransformer(model_name)

    def embed_text(self, text: str) -> np.ndarray:
        return self.model.encode(text, convert_to_numpy=True)

    def embed_documents(self, texts: List[str]) -> np.ndarray:
        return self.model.encode(texts, convert_to_numpy=True)

from typing import List, Dict, Any
from sentence_transformers import CrossEncoder

class CrossEncoderReranker:
    """
    Aday dokümanları sorgu ile yan yana koyarak çapraz sorgulayan yüksek hassasiyetli Reranker.
    """
    def __init__(self, model_name: str = "cross-encoder/ms-marco-MiniLM-L-6-v2"):
        self.model = CrossEncoder(model_name)

    def rerank(self, query: str, candidate_results: List[Dict[str, Any]], top_k: int = 3) -> List[Dict[str, Any]]:
        if not candidate_results:
            return []

        # (Sorgu, Doküman İçeriği) çiftleri oluşturulur
        pairs = [(query, item["document"]["content"]) for item in candidate_results]
        scores = self.model.predict(pairs)

        reranked = []
        for idx, score in enumerate(scores):
            item = candidate_results[idx].copy()
            item["rerank_score"] = float(score)
            reranked.append(item)

        # Rerank skoruna göre büyükten küçüğe sırala
        reranked.sort(key=lambda x: x["rerank_score"], reverse=True)
        
        for rank, item in enumerate(reranked[:top_k], start=1):
            item["final_rank"] = rank

        return reranked[:top_k]

from typing import List, Dict, Any
from rank_bm25 import BM25Okapi

class SparseBM25Search:
    """
    Tam kelime ve terim eşleşmelerini (TRX, USDT, 2FA vb.) yakalamak için BM25 Arama Katmanı.
    """
    def __init__(self, documents: List[Dict[str, Any]]):
        self.documents = documents
        # Basit bir kelime bölme (tokenization) stratejisi
        self.corpus = [doc["content"].lower().split() for doc in documents]
        self.bm25 = BM25Okapi(self.corpus)

    def search(self, query: str, top_k: int = 5) -> List[Dict[str, Any]]:
        tokenized_query = query.lower().split()
        scores = self.bm25.get_scores(tokenized_query)
        
        # Skorlara göre indeksleri sırala
        top_indices = sorted(range(len(scores)), key=lambda i: scores[i], reverse=True)[:top_k]
        
        results = []
        for idx in top_indices:
            results.append({
                "document": self.documents[idx],
                "sparse_score": float(scores[idx]),
                "rank": len(results) + 1
            })
        return results

from typing import List, Dict, Any

class RRFFusion:
    """
    Dense ve Sparse arama sonuçlarını sıralamalarına göre harmanlayan RRF Motoru.
    Formül: RRF_Score = Sum(1 / (k + rank))
    """
    def __init__(self, k: int = 60):
        self.k = k

    def combine_results(
        self, 
        dense_results: List[Dict[str, Any]], 
        sparse_results: List[Dict[str, Any]], 
        top_k: int = 5
    ) -> List[Dict[str, Any]]:
        scores: Dict[str, float] = {}
        doc_map: Dict[str, Dict[str, Any]] = {}

        # Dense sıralamaları ekle
        for item in dense_results:
            doc_id = item["document"]["id"]
            doc_map[doc_id] = item["document"]
            scores[doc_id] = scores.get(doc_id, 0.0) + (1.0 / (self.k + item["rank"]))

        # Sparse (BM25) sıralamaları ekle
        for item in sparse_results:
            doc_id = item["document"]["id"]
            doc_map[doc_id] = item["document"]
            scores[doc_id] = scores.get(doc_id, 0.0) + (1.0 / (self.k + item["rank"]))

        # RRF skoruna göre yeniden sırala
        sorted_ids = sorted(scores.keys(), key=lambda x: scores[x], reverse=True)[:top_k]

        fused_results = []
        for rank, doc_id in enumerate(sorted_ids, start=1):
            fused_results.append({
                "document": doc_map[doc_id],
                "rrf_score": scores[doc_id],
                "rank": rank
            })
        return fused_results

import numpy as np
from typing import List, Dict, Any
from app.core.embedder import DenseEmbedder

class DenseVectorSearch:
    def __init__(self, documents: List[Dict[str, Any]], embedder: DenseEmbedder):
        self.documents = documents
        self.embedder = embedder
        contents = [doc["content"] for doc in documents]
        self.doc_embeddings = self.embedder.embed_documents(contents)

    def search(self, query: str, user_role: str = "GUEST", top_k: int = 5) -> List[Dict[str, Any]]:
        query_embedding = self.embedder.embed_text(query)
        
        norm_query = query_embedding / np.linalg.norm(query_embedding)
        norm_docs = self.doc_embeddings / np.linalg.norm(self.doc_embeddings, axis=1, keepdims=True)
        similarities = np.dot(norm_docs, norm_query)
        
        # Rol Hiyerarşisi (GUEST < ANALYST < ADMIN)
        role_hierarchy = {"GUEST": 1, "ANALYST": 2, "ADMIN": 3}
        user_level = role_hierarchy.get(user_role, 1)

        results = []
        # Skorlara göre sırala
        sorted_indices = np.argsort(similarities)[::-1]
        
        for idx in sorted_indices:
            doc = self.documents[idx]
            doc_required_level = role_hierarchy.get(doc.get("required_role", "GUEST"), 1)
            
            # Metadata Filtering: Kullanıcının yetkisi yetiyorsa sonuçlara ekle
            if user_level >= doc_required_level:
                results.append({
                    "document": doc,
                    "dense_score": float(similarities[idx]),
                    "rank": len(results) + 1
                })
                if len(results) >= top_k:
                    break
        return results

import re

class PIIMasker:
    """
    Kullanıcı sorgularındaki ve dokümanlardaki hassas verileri (PII)
    LLM veya dış servislere sızmaması için maskeleyen güvenlik bileşeni.
    """
    def __init__(self):
        # TCKN (11 haneli), E-posta, Telefon ve Kredi Kartı regex kalıpları
        self.patterns = {
            "TCKN": r'\b[1-9][0-9]{10}\b',
            "EMAIL": r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b',
            "PHONE": r'\b(?:\+?90|0)?\s*[5][0-9]{2}\s*[0-9]{3}\s*[0-9]{2}\s*[0-9]{2}\b',
            "CREDIT_CARD": r'\b(?:\d[ -]*?){13,16}\b'
        }

    def mask(self, text: str) -> str:
        masked_text = text
        for pii_type, pattern in self.patterns.items():
            masked_text = re.sub(pattern, f"[{pii_type}_MASKED]", masked_text)
        return masked_text

from app.data.documents import SAMPLE_DOCUMENTS
from app.core.embedder import DenseEmbedder
from app.core.reranker import CrossEncoderReranker
from app.search.bm25_search import SparseBM25Search
from app.search.vector_search import DenseVectorSearch
from app.search.rrf_fusion import RRFFusion
from app.security.pii_masker import PIIMasker

def run_secured_pipeline(query: str, user_role: str):
    print(f"\n==================================================")
    print(f"Gelen Sorgu: '{query}'")
    print(f"Kullanıcı Rolü: [{user_role}]")
    print(f"==================================================")

    # 1. PII Maskeleme
    pii_masker = PIIMasker()
    clean_query = pii_masker.mask(query)
    print(f"[PII MASKED QUERY]: '{clean_query}'")

    # 2. Servislerin Başlatılması
    embedder = DenseEmbedder()
    vector_search = DenseVectorSearch(SAMPLE_DOCUMENTS, embedder)
    bm25_search = SparseBM25Search(SAMPLE_DOCUMENTS)
    rrf_fusion = RRFFusion(k=60)
    reranker = CrossEncoderReranker()

    # 3. Permission-Aware Vector Search (Metadata Filtering)
    dense_res = vector_search.search(clean_query, user_role=user_role, top_k=5)
    
    # Simple BM25 Filter Strategy for roles
    sparse_res = bm25_search.search(clean_query, top_k=5)
    role_hierarchy = {"GUEST": 1, "ANALYST": 2, "ADMIN": 3}
    filtered_sparse_res = [
        item for item in sparse_res 
        if role_hierarchy.get(user_role, 1) >= role_hierarchy.get(item["document"].get("required_role", "GUEST"), 1)
    ]

    # 4. RRF Fusion & Rerank
    fused_res = rrf_fusion.combine_results(dense_res, filtered_sparse_res, top_k=5)
    final_docs = reranker.rerank(clean_query, fused_res, top_k=2)

    print("\n--- [YETKİLENDİRİLMİŞ VE FİLTRELENMİŞ SONUÇLAR] ---")
    for doc in final_docs:
        print(f"Sıra: {doc['final_rank']} | Score: {doc['rerank_score']:.4f} | Gerekli Rol: [{doc['document']['required_role']}]")
        print(f"Başlık: {doc['document']['title']}")
        print(f"İçerik: {doc['document']['content']}\n")
